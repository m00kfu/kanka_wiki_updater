**Goal:** Add a third "Sync" tab to the web-based review UI that lets users run `sync_pipeline` from the browser with real-time output streaming via Server-Sent Events (SSE).

**Architecture:** Backend spawns sync_pipeline as an unbuffered subprocess, reads stdout line-by-line in a background thread, and streams output to the frontend via SSE. Frontend uses native EventSource API to receive chunks progressively in a terminal-like output area.

**Tech Stack:** Flask (already installed), Python `subprocess` + `threading`, browser-native `EventSource` API

```python
def _sync_thread(job_id, proc, buffer):
    """Background thread that reads subprocess stdout and pushes to deque."""
    try:
        for line in proc.stdout:
            buffer.append(line.rstrip('\n'))
        proc.wait()
        if proc.returncode == 0:
            _sync_jobs[job_id]['status'] = 'completed'
        else:
            _sync_jobs[job_id]['status'] = 'error'
    except Exception as e:
        buffer.append(f'Sync error: {e}')
        _sync_jobs[job_id]['status'] = 'error'
```

```python
            cwd=module_dir,

            bufsize=1,
            universal_newlines=True,

        buffer = deque(maxlen=500)
        _sync_jobs[job_id] = {
            'process': proc,
            'buffer': buffer,
            'status': 'running',
            'started_at': time.time(),
            'finished_at': None,
        }
        thread = threading.Thread(target=_sync_thread, args=(job_id, proc, buffer), daemon=True)
        thread.start()
```

```python
    @app.route('/api/sync/output')
    def sync_output():
        job_id = request.args.get('job_id')
        if not job_id or job_id not in _sync_jobs:
            from flask import Response
            return Response('Job not found', status=404, mimetype='text/plain')

        job = _sync_jobs[job_id]
        buffer = job['buffer']
        last_flush = [0]  # mutable index into buffer

        def generate():
            while True:
                if job['status'] in ('completed', 'error'):
                    # Drain remaining lines
                    while buffer:
                        yield f'event: message\ndata: {json.dumps({"type": "output", "text": buffer.popleft()})}\n\n'
                    yield f'event: status\ndata: {json.dumps({"status": job["status"]})}\n\n'
                    yield 'event: end\n\n'
                    break

                # Yield any new lines from buffer
                while buffer:
                    line = buffer.popleft()
                    yield f'event: message\ndata: {json.dumps({"type": "output", "text": line})}\n\n'

                # Check status again after draining
                if job['status'] in ('completed', 'error'):
                    yield f'event: status\ndata: {json.dumps({"status": job["status"]})}\n\n'
                    yield 'event: end\n\n'
                    break

                time.sleep(0.2)  # poll interval

        from flask import Response
        return Response(generate(), mimetype='text/event-stream')
```

```javascript
async function runSync() {
    if (currentSyncJob && currentSyncJob.status === 'running') {
        cancelSync();
        return;
    }

    var result = await apiCall('/api/sync/run', 'POST');
    if (!result) return;

    currentSyncJob = {job_id: result.job_id, status: 'running', output: ''};

    // Connect to SSE stream
    var eventSource = new EventSource('/api/sync/output?job_id=' + result.job_id);
    currentSyncJob.eventSource = eventSource;

    eventSource.addEventListener('message', function(e) {
        var data = JSON.parse(e.data);
        if (data.type === 'output') {
            currentSyncJob.output += data.text + '\n';
            renderContent();
            // Auto-scroll output area
            if (syncOutputEl) syncOutputEl.scrollTop = syncOutputEl.scrollHeight;
        }
    });

    eventSource.addEventListener('status', function(e) {
        var data = JSON.parse(e.data);
        currentSyncJob.status = data.status;
        renderContent();
    });

    eventSource.addEventListener('end', function() {
        eventSource.close();
        // Refresh proposal list after sync completes
        refreshProposals();
        showToast('Sync completed: ' + currentSyncJob.status, currentSyncJob.status === 'completed' ? 'success' : 'error');
        currentSyncJob = null;
    });

    renderContent();
}

function cancelSync() {
    if (currentSyncJob && currentSyncJob.eventSource) {
        fetch('/api/sync/cancel?job_id=' + currentSyncJob.job_id, {method: 'POST'});
        currentSyncJob.eventSource.close();
        currentSyncJob = null;
        renderContent();
        showToast('Sync cancelled', 'error');
    }
}

function refreshProposals() {
    fetch('/api/proposals')
      .then(function(r) { return r.json(); })
      .then(function(data) { proposals = data; updateStats(); renderSidebar(); });
}
```

```python
    @app.route('/api/sync/cancel', methods=['POST'])
    def sync_cancel():
        job_id = request.args.get('job_id')
        if not job_id or job_id not in _sync_jobs:
            return jsonify({'error': 'Job not found'}), 404

        job = _sync_jobs[job_id]
        proc = job['process']
        if proc and proc.poll() is None:
            proc.terminate()
            try:
                proc.wait(timeout=5)
            except subprocess.TimeoutExpired:
                proc.kill()
        _sync_jobs[job_id]['status'] = 'cancelled'
        return jsonify({'ok': True})
```
